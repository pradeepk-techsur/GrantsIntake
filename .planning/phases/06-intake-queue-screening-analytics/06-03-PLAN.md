**1c. `getReviewHandoff(entryId, grantorOrgId)` — public method**

```typescript
async getReviewHandoff(
  entryId: string,
  grantorOrgId: string,
): Promise<Record<string, unknown> | null> {
  // IDOR guard: verify entry belongs to this grantor
  const entryResult = await pool.query(
    `SELECT iqe.entry_id, go.org_id AS grantor_org_id
     FROM intake_queue_entries iqe
     JOIN opportunities opp ON opp.opportunity_id = iqe.opportunity_id
     JOIN programs p ON p.program_id = opp.program_id
     JOIN grantor_organizations go ON go.org_id = p.grantor_org_id
     WHERE iqe.entry_id = $1`,
    [entryId],
  );

  if (entryResult.rowCount === 0) {
    const err = new Error('Queue entry not found') as Error & { status: number };
    err.status = 404;
    throw err;
  }

  if (entryResult.rows[0].grantor_org_id !== grantorOrgId) {
    const err = new Error('Access denied') as Error & { status: number; code: string };
    err.status = 403;
    err.code = 'PERMISSION_DENIED';
    throw err;
  }

  const result = await pool.query(
    `SELECT handoff_id, entry_id, snapshot_id, review_workflow_type, assigned_reviewer_ids, handed_off_at, created_by
     FROM review_handoffs
     WHERE entry_id = $1
     ORDER BY handed_off_at DESC
     LIMIT 1`,
    [entryId],
  );

  return result.rows[0] ?? null;
}
```

**1d. `getIntakeAnalytics(grantorOrgId)` — public method**

```typescript
async getIntakeAnalytics(
  grantorOrgId: string,
): Promise<{ by_status: { status: string; count: number }[]; total: number }> {
  const result = await pool.query<{ status: string; count: string }>(
    `SELECT iqe.status, COUNT(*)::text AS count
     FROM intake_queue_entries iqe
     JOIN opportunities opp ON opp.opportunity_id = iqe.opportunity_id
     JOIN programs p ON p.program_id = opp.program_id
     JOIN grantor_organizations go ON go.org_id = p.grantor_org_id
     WHERE go.org_id = $1
     GROUP BY iqe.status
     ORDER BY count DESC`,
    [grantorOrgId],
  );

  const by_status = result.rows.map((r) => ({
    status: r.status,
    count: parseInt(r.count, 10),
  }));

  const total = by_status.reduce((sum, r) => sum + r.count, 0);

  return { by_status, total };
}
```

**1e. `exportIntakeCsv(filters, grantorOrgId)` — public method**

```typescript
async exportIntakeCsv(
  filters: Omit<QueueFilters, 'page' | 'page_size'>,
  grantorOrgId: string,
): Promise<string> {
  // Build WHERE clauses (same IDOR guard as getQueueEntries)
  const conditions: string[] = ['go.org_id = $1'];
  const params: unknown[] = [grantorOrgId];
  let paramIdx = 2;

  if (filters.opportunity_id) {
    conditions.push(`iqe.opportunity_id = $${paramIdx}`);
    params.push(filters.opportunity_id);
    paramIdx++;
  }
  if (filters.status) {
    conditions.push(`iqe.status = $${paramIdx}`);
    params.push(filters.status);
    paramIdx++;
  }

  const whereClause = conditions.join(' AND ');

  const result = await pool.query(
    `SELECT
       iqe.entry_id,
       iqe.status,
       org.legal_name AS org_name,
       opp.title AS opportunity_title,
       ss.confirmation_number,
       ss.submitted_at AS submission_timestamp,
       (ss.budget_snapshot->>'total_federal_requested')::numeric AS requested_amount,
       COALESCE(jsonb_array_length(ss.attachment_refs), 0) AS attachment_count,
       ss.eligibility_snapshot->>'overall_result' AS eligibility_result,
       iqe.created_at
     FROM intake_queue_entries iqe
     JOIN submission_snapshots ss ON ss.snapshot_id = iqe.snapshot_id
     JOIN organizations org ON org.org_id = iqe.org_id
     JOIN opportunities opp ON opp.opportunity_id = iqe.opportunity_id
     JOIN programs p ON p.program_id = opp.program_id
     JOIN grantor_organizations go ON go.org_id = p.grantor_org_id
     WHERE ${whereClause}
     ORDER BY ss.submitted_at DESC`,
    params,
  );

  // Build CSV string
  const headers = [
    'entry_id', 'status', 'org_name', 'opportunity_title',
    'confirmation_number', 'submission_timestamp', 'requested_amount',
    'attachment_count', 'eligibility_result', 'created_at',
  ];

  const escapeCsv = (val: unknown): string => {
    if (val === null || val === undefined) return '';
    const str = String(val);
    // Wrap in quotes if contains comma, quote, or newline
    if (str.includes(',') || str.includes('"') || str.includes('\n')) {
      return `"${str.replace(/"/g, '""')}"`;
    }
    return str;
  };

  const rows = result.rows.map((row) =>
    headers.map((h) => escapeCsv(row[h])).join(','),
  );

  return [headers.join(','), ...rows].join('\n');
}
```

import { useQuery } from '@tanstack/react-query';
import { Link } from 'react-router-dom';
import { workspaceApi } from '../../api/workspaceApi';
import { intakeQueueApi } from '../../api/intakeQueueApi';
import type { Workspace } from '../../types/workspace';

export function ApplicantDashboard() {
  const {
    data: workspaces = [],
    isLoading: workspacesLoading,
    isError: workspacesError,
  } = useQuery<Workspace[]>({
    queryKey: ['workspaces'],
    queryFn: workspaceApi.listWorkspaces,
    staleTime: 30000,
  });

  const { data: notificationsData } = useQuery({
    queryKey: ['notifications', { is_read: false }],
    queryFn: () => intakeQueueApi.getNotifications({ is_read: false }).then((r) => r.data),
    staleTime: 30000,
  });

  const unreadCount = notificationsData?.total ?? 0;

  // Partition workspaces into in-progress and submitted
  const inProgress = workspaces.filter((w) => !['submitted', 'intake_screening', 'accepted_for_review', 'administratively_rejected', 'withdrawn'].includes(w.status));
  const submitted = workspaces.filter((w) => ['submitted', 'intake_screening', 'accepted_for_review', 'returned_for_correction', 'resubmitted'].includes(w.status));

  if (workspacesLoading) {
    return (
      <div aria-busy="true" aria-label="Loading dashboard">
        <p>Loading your dashboard…</p>
      </div>
    );
  }

  if (workspacesError) {
    return (
      <div className="usa-alert usa-alert--error" role="alert">
        <div className="usa-alert__body">
          <p className="usa-alert__text">Failed to load dashboard data. Please refresh.</p>
        </div>
      </div>
    );
  }

  return (
    <div data-testid="applicant-dashboard">
      <h1 className="usa-prose">My Dashboard</h1>

      {/* Unread notifications badge */}
      {unreadCount > 0 && (
        <div className="usa-alert usa-alert--info margin-bottom-2" data-testid="unread-notifications-alert">
          <div className="usa-alert__body">
            <p className="usa-alert__text">
              You have {unreadCount} unread notification{unreadCount > 1 ? 's' : ''}.{' '}
              <Link to="/applicant/notifications">View notifications</Link>
            </p>
          </div>
        </div>
      )}

      {/* Application Progress */}
      <h2 className="usa-prose" data-testid="section-in-progress">Applications In Progress</h2>
      {inProgress.length === 0 ? (
        <div className="usa-alert usa-alert--info" data-testid="no-inprogress">
          <div className="usa-alert__body">
            <p className="usa-alert__text">
              No applications in progress.{' '}
              <Link to="/opportunities">Find an opportunity</Link> to get started.
            </p>
          </div>
        </div>
      ) : (
        <ul className="usa-card-group" data-testid="inprogress-list">
          {inProgress.map((workspace) => (
            <li key={workspace.workspace_id} className="usa-card tablet:grid-col-6" data-testid="workspace-card">
              <div className="usa-card__container">
                <div className="usa-card__header">
                  <h3 className="usa-card__heading">{workspace.opportunity_id}</h3>
                </div>
                <div className="usa-card__body">
                  <p>Status: <strong>{workspace.status.replace(/_/g, ' ')}</strong></p>
                  <p>Started: {new Date(workspace.created_at).toLocaleDateString()}</p>
                </div>
                <div className="usa-card__footer">
                  <Link
                    to={`/applicant/workspaces/${workspace.workspace_id}`}
                    className="usa-button"
                    data-testid="continue-application-link"
                  >
                    Continue Application
                  </Link>
                </div>
              </div>
            </li>
          ))}
        </ul>
      )}

      {/* Submission History */}
      <h2 className="usa-prose margin-top-4" data-testid="section-submitted">Submission History</h2>
      {submitted.length === 0 ? (
        <div className="usa-alert usa-alert--info" data-testid="no-submissions">
          <div className="usa-alert__body">
            <p className="usa-alert__text">No submitted applications yet.</p>
          </div>
        </div>
      ) : (
        <table className="usa-table usa-table--borderless width-full" data-testid="submissions-table">
          <thead>
            <tr>
              <th scope="col">Opportunity</th>
              <th scope="col">Status</th>
              <th scope="col">Submitted</th>
              <th scope="col">Actions</th>
            </tr>
          </thead>
          <tbody>
            {submitted.map((workspace) => (
              <tr key={workspace.workspace_id} data-testid="submission-row">
                <td>{workspace.opportunity_id}</td>
                <td><span className="usa-tag">{workspace.status.replace(/_/g, ' ')}</span></td>
                <td>{new Date(workspace.updated_at).toLocaleDateString()}</td>
                <td>
                  <Link

                    data-testid="view-receipt-link"
                  >
                    View Receipt

      )}
    </div>
  );
}

| Boundary | Description |
|----------|-------------|
| client→API | Untrusted query params (`opportunity_id`, `status`, `format`) crossing into the export and analytics handlers |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-06-03-01 | Information disclosure / Elevation | GET /intake-analytics — IDOR: grantor A reading analytics for grantor B's opportunities | mitigate | `intakeQueueService.getIntakeAnalytics()` JOINs through `opportunities → programs → grantor_organizations` and filters `WHERE go.org_id = $1` where `grantorOrgId` is server-derived via `getGrantorOrgIdForUser()` in `src/routes/intakeQueue.ts`. Never uses client-supplied org ID. |
| T-06-03-02 | Information disclosure / Elevation | GET /intake-queue/export — IDOR: grantor A downloading CSV containing grantor B's application data | mitigate | `intakeQueueService.exportIntakeCsv()` applies the same `WHERE go.org_id = $1` JOIN guard as `getQueueEntries()`. | T-06-03-03 | Tampering | GET /intake-queue/export path collision — 'export' literal matching `:entryId` param in `/intake-queue/:entryId` | mitigate | Export route is registered BEFORE the parameterized `/intake-queue/:entryId` route in `src/routes/intakeQueue.ts`. Express evaluates routes in registration order; the literal path `/intake-queue/export` wins. Documented in route file comment and verified by integration test 7. |
