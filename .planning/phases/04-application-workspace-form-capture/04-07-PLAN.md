Fix both:

1. Add `import { useMutation } from '@tanstack/react-query';` and `import { useNavigate } from 'react-router-dom';` (extend existing react-router-dom import to include useNavigate).

2. Add `import { workspaceApi } from '../../api/workspaceApi';` near the top.

3. Inside the component, add a `useNavigate()` call and a `useMutation` for workspace creation:
```tsx
const navigate = useNavigate();
const { user } = useAuthStore();

const createWorkspaceMutation = useMutation({
  mutationFn: () =>
    workspaceApi.createWorkspace({
      opportunity_id: opportunity!.opportunity_id,
    }),
  onSuccess: (data) => {
    navigate(`/applicant/workspaces/${data.workspace.workspace_id}`);
  },
  onError: (err: unknown) => {
    // 409 DUPLICATE_WORKSPACE: the workspace already exists; fetch its id and navigate
    const anyErr = err as { response?: { data?: { error_code?: string; workspace_id?: string } } };
    if (anyErr?.response?.data?.error_code === 'DUPLICATE_WORKSPACE' && anyErr?.response?.data?.workspace_id) {
      navigate(`/applicant/workspaces/${anyErr.response.data.workspace_id}`);
    }
  },
});
```

4. In the `renderCTA` function, replace the "Start Application" `<a>` with a button:
```tsx
return (
  <button
    type="button"
    className="usa-button usa-button--big"
    style={{ width: '100%' }}
    onClick={() => createWorkspaceMutation.mutate()}
    disabled={createWorkspaceMutation.isPending}
    data-testid="start-application-btn"
  >
    {createWorkspaceMutation.isPending ? 'Starting…' : 'Start Application'}
```
