CREATE TABLE application_workspaces (
    workspace_id        UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    opportunity_id      UUID NOT NULL REFERENCES opportunities(opportunity_id),
    org_id              UUID NOT NULL REFERENCES organizations(org_id),
    track_id            UUID,
    status              VARCHAR(50) NOT NULL DEFAULT 'workspace_created',
    -- workspace_created, in_progress, ready_for_internal_review,
    -- ready_to_submit, submitted, intake_screening, returned_for_correction,
    -- resubmitted, accepted_for_review, withdrawn, administratively_rejected
    visibility          VARCHAR(20) NOT NULL DEFAULT 'grantee_private',
    -- grantee_private (draft) | shared (submitted)
    is_locked           BOOLEAN NOT NULL DEFAULT FALSE,
    created_by          UUID NOT NULL REFERENCES users(user_id),
    created_at          TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT uq_workspace_org_opp UNIQUE (opportunity_id, org_id)
    -- Note: constraint conditionally bypassed when opportunity.duplicate_allowed = true
);
CREATE INDEX idx_workspaces_opportunity ON application_workspaces(opportunity_id);
CREATE INDEX idx_workspaces_org ON application_workspaces(org_id);
CREATE INDEX idx_workspaces_status ON application_workspaces(status);

CREATE TABLE application_sections (
    section_id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id        UUID NOT NULL REFERENCES application_workspaces(workspace_id),
    section_type        VARCHAR(50) NOT NULL,
    -- org_profile, eligibility, narrative, budget, workplan, performance_measures,
    -- attachments, certifications, review_submit, custom
    section_name        VARCHAR(250) NOT NULL,
    status              VARCHAR(20) NOT NULL DEFAULT 'not_started',
    -- not_started, in_progress, complete, error, locked
    is_visible          BOOLEAN NOT NULL DEFAULT TRUE,
    is_locked           BOOLEAN NOT NULL DEFAULT FALSE,
    display_order       INTEGER NOT NULL DEFAULT 0,
    owner_id            UUID REFERENCES users(user_id),
    internal_due_date   DATE,
    validation_status   VARCHAR(20) DEFAULT 'not_validated',
    validation_errors   JSONB,
    -- array of {field_id, severity: blocking|warning|info, message, field_label}
    visibility          VARCHAR(20) NOT NULL DEFAULT 'grantee_private',
    created_at          TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_sections_workspace ON application_sections(workspace_id);
CREATE INDEX idx_sections_type ON application_sections(workspace_id, section_type);

CREATE TABLE workspace_tasks (
    task_id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id    UUID NOT NULL REFERENCES application_workspaces(workspace_id),
    section_id      UUID REFERENCES application_sections(section_id),
    task_title      VARCHAR(500) NOT NULL,
    assignee_id     UUID NOT NULL REFERENCES users(user_id),
    task_due_date   DATE,
    task_notes      TEXT,
    status          VARCHAR(20) NOT NULL DEFAULT 'open',  -- open, complete
    created_by      UUID NOT NULL REFERENCES users(user_id),
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    completed_at    TIMESTAMPTZ
);
CREATE INDEX idx_tasks_workspace ON workspace_tasks(workspace_id);
CREATE INDEX idx_tasks_assignee ON workspace_tasks(assignee_id);

-- CRITICAL: workspace_comments are NEVER visible to grantor roles
CREATE TABLE workspace_comments (
    comment_id      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id    UUID NOT NULL REFERENCES application_workspaces(workspace_id),
    section_id      UUID REFERENCES application_sections(section_id),
    comment_text    TEXT NOT NULL CHECK (char_length(comment_text) <= 5000),
    visibility      VARCHAR(20) NOT NULL DEFAULT 'internal',
    posted_by       UUID NOT NULL REFERENCES users(user_id),
    posted_at       TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_comments_workspace ON workspace_comments(workspace_id);

-- Add workspace_id FK to eligibility_responses (deferred from Phase 3 migration 011)
ALTER TABLE eligibility_responses
    ADD COLUMN IF NOT EXISTS workspace_id_fk UUID REFERENCES application_workspaces(workspace_id);

**Step 4: Create src/routes/workspaces.ts**

Import authenticate, requireRole, workspaceService, GRANTOR_ROLES, UUID_REGEX pattern. Mount all endpoints:

CRITICAL SECURITY RULES for this router:
1. **Two-step IDOR guard on all `:id` routes**: First check workspace EXISTS (404), then check `workspaceService.verifyWorkspaceMember(workspaceId, req.user.user_id)` (403). This prevents org enumeration.
2. **Comments routes PERMANENTLY block grantor roles at router layer** (not just middleware): Check if `GRANTOR_ROLES.some(r => req.user!.roles.includes(r))` → 403 GRANTOR_ACCESS_DENIED. This is required by PRD-INTAKE-033 / T-04-03.
3. **org_id derived server-side**: The POST /workspaces endpoint calls `organizationService.getOrgIdForUser(userId)`, never trusts request body for org_id.
4. **UUID format guards** (UUID_REGEX) on all :id, :sectionId, :taskId params → 404 if not valid UUID.

Endpoints:
- `POST /api/v1/workspaces` — authenticate; derive org_id server-side; 409 on duplicate (DUPLICATE_WORKSPACE); 201 on success
- `GET /api/v1/workspaces` — authenticate; list workspaces for current user's org
- `GET /api/v1/workspaces/:id` — authenticate + IDOR guard
- `GET /api/v1/workspaces/:id/sections` — authenticate + IDOR guard; returns sections array with is_visible filter applied
- `GET /api/v1/workspaces/:id/sections/:sectionId` — authenticate + IDOR guard
- `PUT /api/v1/workspaces/:id/sections/:sectionId/assignment` — authenticate + IDOR guard; validate req.user has org_roles with role IN ('proposal_lead', 'org_admin') on this workspace's org_id → 403 if not; Zod: { owner_id?: uuid, internal_due_date?: date-string }
- `GET /api/v1/workspaces/:id/tasks` — authenticate + IDOR guard
- `POST /api/v1/workspaces/:id/tasks` — authenticate + IDOR guard; Zod: { section_id?, task_title, assignee_id, task_due_date?, task_notes? }
- `PUT /api/v1/workspaces/:id/tasks/:taskId` — authenticate + IDOR guard; Zod: { task_title?, status?, task_due_date?, task_notes? }
- `DELETE /api/v1/workspaces/:id/tasks/:taskId` — authenticate + IDOR guard; 204
- `GET /api/v1/workspaces/:id/comments` — authenticate + GRANTOR_BLOCK + IDOR guard
- `POST /api/v1/workspaces/:id/comments` — authenticate + GRANTOR_BLOCK + IDOR guard; Zod: { section_id?, comment_text: string max 5000 chars }

import apiClient from './client';
import type { Workspace, WorkspaceSection, WorkspaceTask, WorkspaceComment,
  CreateWorkspaceInput, AssignSectionInput, CreateTaskInput, CreateCommentInput } from '../types/workspace';

export const workspaceApi = {
  createWorkspace: (input: CreateWorkspaceInput) =>
    apiClient.post<{ workspace: Workspace; sections: WorkspaceSection[] }>('/workspaces', input).then(r => r.data),
  listWorkspaces: () =>
    apiClient.get<Workspace[]>('/workspaces').then(r => r.data),
  getWorkspace: (workspaceId: string) =>
    apiClient.get<Workspace>(`/workspaces/${workspaceId}`).then(r => r.data),
  getSections: (workspaceId: string) =>
    apiClient.get<WorkspaceSection[]>(`/workspaces/${workspaceId}/sections`).then(r => r.data),
  getSection: (workspaceId: string, sectionId: string) =>
    apiClient.get<WorkspaceSection>(`/workspaces/${workspaceId}/sections/${sectionId}`).then(r => r.data),
  assignSection: (workspaceId: string, sectionId: string, input: AssignSectionInput) =>
    apiClient.put(`/workspaces/${workspaceId}/sections/${sectionId}/assignment`, input).then(r => r.data),
  getTasks: (workspaceId: string) =>
    apiClient.get<WorkspaceTask[]>(`/workspaces/${workspaceId}/tasks`).then(r => r.data),
  createTask: (workspaceId: string, input: CreateTaskInput) =>
    apiClient.post<WorkspaceTask>(`/workspaces/${workspaceId}/tasks`, input).then(r => r.data),
  updateTask: (workspaceId: string, taskId: string, updates: Partial<WorkspaceTask>) =>
    apiClient.put(`/workspaces/${workspaceId}/tasks/${taskId}`, updates).then(r => r.data),
  deleteTask: (workspaceId: string, taskId: string) =>
    apiClient.delete(`/workspaces/${workspaceId}/tasks/${taskId}`),
  getComments: (workspaceId: string) =>
    apiClient.get<WorkspaceComment[]>(`/workspaces/${workspaceId}/comments`).then(r => r.data),
  postComment: (workspaceId: string, input: CreateCommentInput) =>
    apiClient.post<WorkspaceComment>(`/workspaces/${workspaceId}/comments`, input).then(r => r.data),
};
```

// Layout: two-column — left sidebar (WorkspaceSidebar, ~280px) + main content (WorkspaceSectionPanel)
// On mount: load workspace + sections via React Query
// useWorkspaceStore for activeSectionType (default to first visible section)

const { workspaceId } = useParams<{ workspaceId: string }>();

// React Query
const workspaceQuery = useQuery(['workspace', workspaceId], () => workspaceApi.getWorkspace(workspaceId!));
const sectionsQuery = useQuery(['sections', workspaceId], () => workspaceApi.getSections(workspaceId!));

// Initialize active section to first visible section on load
const { activeSectionType, setActiveSectionType } = useWorkspaceStore();
useEffect(() => {
  if (sectionsQuery.data && !activeSectionType) {
    const first = sectionsQuery.data.find(s => s.is_visible);
    if (first) setActiveSectionType(first.section_type);
  }
}, [sectionsQuery.data, activeSectionType, setActiveSectionType]);

const activeSection = sectionsQuery.data?.find(s => s.section_type === activeSectionType);
```

|----------|-------------|
| client→API | Untrusted request body / params crossing into workspace route handlers |
| org_membership→workspace | User-supplied workspace_id used to gate workspace access before org membership verified |
| commenter→workspace | Applicant POSTing comment; must be blocked for all grantor roles at router layer (not just middleware) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01 | Elevation of privilege | POST /workspaces — org_id from body | mitigate | org_id derived exclusively via `organizationService.getOrgIdForUser(req.user.user_id)` in workspaceService.createWorkspace; request body org_id field is rejected (Zod schema does not accept it) |
| T-04-02 | Information disclosure | GET /workspaces/:id — IDOR | mitigate | Two-step guard in workspacesRouter: (1) check workspace EXISTS → 404 if not; (2) check `workspaceService.verifyWorkspaceMember(workspaceId, req.user.user_id)` → 403 if false; mirrors T-03-22 pattern from organizationsRouter |
| T-04-03 | Information disclosure | GET/POST /workspaces/:id/comments — grantor access | mitigate | Grantor block at router layer before IDOR guard: `if (GRANTOR_ROLES.some(r => req.user!.roles.includes(r))) return res.status(403).json({ error: 'GRANTOR_ACCESS_DENIED' })` — hardcoded in workspacesRouter, cannot be bypassed by middleware ordering |
| T-04-04 | Tampering | PUT /workspaces/:id/sections/:sectionId/assignment — role escalation | mitigate | Route checks org_roles for the workspace's org_id: only org_admin or proposal_lead may assign sections; check performed in workspacesRouter before calling workspaceService.assignSection |
| T-04-05 | Spoofing | Bearer token validation | accept | Delegated to authenticate middleware (jose verifyAccessToken) — already mitigated in Phase 1 (T-02-04); residual risk owned by auth service |
| T-04-06 | Denial of service | POST /workspaces/:id/comments — 5000-char comment flood | mitigate | DB-level CHECK constraint `char_length(comment_text) <= 5000` in migration 012 + Zod max(5000) in route schema |
