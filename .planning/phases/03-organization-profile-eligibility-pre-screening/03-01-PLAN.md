```sql
-- Migration 010: organizations, org_contacts, org_roles, org_attachments
-- (applicant org profile schema)

CREATE TABLE organizations (
    org_id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    legal_name              VARCHAR(250) NOT NULL,
    dba_name                VARCHAR(250),
    address_line1           VARCHAR(250) NOT NULL,
    address_line2           VARCHAR(250),
    city                    VARCHAR(100) NOT NULL,
    state                   CHAR(2) NOT NULL,
    zip                     VARCHAR(10) NOT NULL,
    country                 CHAR(2) NOT NULL DEFAULT 'US',
    entity_type             VARCHAR(50) NOT NULL,
    -- nonprofit_501c3, nonprofit_other, for_profit, government_federal,
    -- government_state, government_local, tribal, university, individual, other
    ein                     CHAR(9),
    uei                     CHAR(12),
    sam_registered          BOOLEAN NOT NULL DEFAULT FALSE,
    sam_expiration_date     DATE,
    tax_exempt_status       VARCHAR(20),
    -- 501c3, 501c4, 501c6, other, not_applicable
    congressional_district  VARCHAR(20),
    primary_contact_name    VARCHAR(250) NOT NULL,
    primary_contact_email   VARCHAR(320) NOT NULL,
    primary_contact_phone   VARCHAR(30),
    banking_readiness       VARCHAR(20) NOT NULL DEFAULT 'unknown',
    -- ready, not_ready, unknown
    indirect_cost_rate      NUMERIC(5,2),
    indirect_cost_base      VARCHAR(100),
    profile_completeness_pct NUMERIC(5,2) DEFAULT 0,
    created_at              TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at              TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT chk_uei_format CHECK (uei IS NULL OR uei ~ '^[A-Za-z0-9]{12}$'),
    CONSTRAINT chk_ein_format CHECK (ein IS NULL OR ein ~ '^\d{9}$')
);
CREATE INDEX idx_organizations_uei ON organizations(uei);
CREATE INDEX idx_organizations_ein ON organizations(ein);

CREATE TABLE org_contacts (
    contact_id      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    org_id          UUID NOT NULL REFERENCES organizations(org_id),
    user_id         UUID REFERENCES users(user_id),
    contact_name    VARCHAR(250) NOT NULL,
    contact_email   VARCHAR(320) NOT NULL,
    contact_phone   VARCHAR(30),
    contact_title   VARCHAR(250),
    contact_type    VARCHAR(50) NOT NULL,
    -- primary, authorized_representative, financial, technical, other
    is_active       BOOLEAN NOT NULL DEFAULT TRUE,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_org_contacts_org ON org_contacts(org_id);

CREATE TABLE org_roles (
    role_id                 UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    org_id                  UUID NOT NULL REFERENCES organizations(org_id),
    user_id                 UUID NOT NULL REFERENCES users(user_id),
    roles                   JSONB NOT NULL,
    -- array of: org_admin, proposal_lead, contributor, finance_contributor,
    --           authorized_representative, external_contributor
    invited_by              UUID REFERENCES users(user_id),
    invitation_sent_at      TIMESTAMPTZ,
    invitation_accepted_at  TIMESTAMPTZ,
    created_at              TIMESTAMPTZ NOT NULL DEFAULT now(),
    revoked_at              TIMESTAMPTZ,
    CONSTRAINT uq_org_user_role UNIQUE (org_id, user_id)
);
CREATE INDEX idx_org_roles_org ON org_roles(org_id);
CREATE INDEX idx_org_roles_user ON org_roles(user_id);

CREATE TABLE org_attachments (
    attachment_id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    org_id                  UUID NOT NULL REFERENCES organizations(org_id),
    document_type           VARCHAR(100) NOT NULL,
    -- irs_determination_letter, w9, audit_report, indirect_cost_agreement,
    -- board_roster, insurance_certificate, letters_of_support, other
    custom_document_name    VARCHAR(250),
    version_number          INTEGER NOT NULL DEFAULT 1,
    file_name               VARCHAR(500) NOT NULL,
    file_path               VARCHAR(2048) NOT NULL,  -- S3 object key
    mime_type               VARCHAR(100) NOT NULL,
    file_size_bytes         BIGINT NOT NULL,
    expiration_date         DATE,
    is_active               BOOLEAN NOT NULL DEFAULT TRUE,
    uploaded_by             UUID NOT NULL REFERENCES users(user_id),
    uploaded_at             TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT chk_version_positive CHECK (version_number > 0)
);
CREATE INDEX idx_org_attachments_org ON org_attachments(org_id);
CREATE INDEX idx_org_attachments_type ON org_attachments(org_id, document_type);
CREATE INDEX idx_org_attachments_active ON org_attachments(org_id, document_type, is_active);
```

**Step C: Create `src/services/organization/organizationService.ts`**

Implement the following class methods using `pool.query` (same pattern as eligibilityService.ts):

1. **`computeCompleteness(org: Partial<Organization>): number`** — returns percentage of filled required fields:
   Required fields: legal_name, address_line1, city, state, zip, entity_type, primary_contact_name, primary_contact_email, banking_readiness, ein, uei, sam_registered (12 total, each worth 1/12 × 100). Return rounded to 2 decimal places.

2. **`createOrg(userId: string, input: CreateOrgInput): Promise<Organization>`** — INSERT into organizations, set profile_completeness_pct, then INSERT into org_roles with roles=['org_admin'] for the creator. Then INSERT into audit_events with:
   ```sql
   INSERT INTO audit_events (actor_user_id, action, entity_type, entity_id, details)
   VALUES ($1, 'ORGANIZATION_PROFILE_CREATED', 'organization', $2, $3::jsonb)
   ```
   Return the created organization row.

3. **`getOrg(orgId: string): Promise<Organization | null>`** — SELECT from organizations WHERE org_id = $1. Return null if not found.

4. **`getOrgIdForUser(userId: string): Promise<string | null>`** — SELECT org_id FROM org_roles WHERE user_id = $1 AND revoked_at IS NULL LIMIT 1. Returns null if user has no org.

5. **`verifyOrgMember(orgId: string, userId: string): Promise<boolean>`** — SELECT 1 FROM org_roles WHERE org_id = $1 AND user_id = $2 AND revoked_at IS NULL. Returns true if member.

6. **`verifyOrgAdmin(orgId: string, userId: string): Promise<boolean>`** — SELECT 1 FROM org_roles WHERE org_id = $1 AND user_id = $2 AND revoked_at IS NULL AND roles @> '["org_admin"]'::jsonb.

7. **`updateOrg(orgId: string, callerId: string, input: UpdateOrgInput): Promise<Organization>`** — verify caller is org_admin (throw 403 AppError if not). UPDATE organizations SET (all provided fields), recompute profile_completeness_pct, set updated_at = now(). Emit ORGANIZATION_PROFILE_UPDATED audit event. Return updated row.

8. **`getCredentialStatus(orgId: string, warningWindowDays = 60): Promise<CredentialStatus>`** — Check these credential items:
   - `sam_expiration` — from organizations.sam_expiration_date
   - Active org_attachments where expiration_date IS NOT NULL for types: irs_determination_letter, w9, audit_report, indirect_cost_agreement, insurance_certificate
   
   For each item compute:
   - `days_remaining = differenceInDays(expiration_date, today)` (can be negative if past)
   - `status`: expired if days_remaining < 0, expiring_soon if 0 <= days_remaining <= warningWindowDays, valid otherwise
   - Items with no expiration_date are not included in the list.

9. **`listDocuments(orgId: string): Promise<OrgDocument[]>`** — SELECT from org_attachments WHERE org_id = $1 AND is_active = true ORDER BY document_type, version_number DESC. Compute expiration_status for each (same logic as getCredentialStatus).

10. **`uploadDocument(orgId: string, callerId: string, meta: UploadDocumentMeta, fileBuffer: Buffer): Promise<OrgDocument>`** — Compute version_number by getting MAX(version_number) for same org_id + document_type + is_active=true, add 1 (default 1 if none). INSERT into org_attachments. For file_path use S3 key format: `orgs/{orgId}/docs/{documentType}/{Date.now()}-{fileName}`. Use the existing S3 client from the project (check docker-compose for S3-compatible storage — upload using `@aws-sdk/client-s3` if available, otherwise store file_path as a local path under `uploads/orgs/`). Return inserted row with computed expiration_status.

11. **`listDocumentVersions(orgId: string, docId: string): Promise<OrgDocument[]>`** — Find the document_type for docId, then SELECT all versions (both active and inactive) for that org_id + document_type ORDER BY version_number DESC.

Implement endpoints:

**POST /api/v1/organizations** — authenticate. Parse createOrgSchema. Call organizationService.createOrg(req.user.user_id, input). Return 201 + org object.

**GET /api/v1/organizations/:org_id** — authenticate. UUID guard: if org_id doesn't match UUID regex, return 404. Call organizationService.getOrg(org_id). Return 404 if null. IDOR: call organizationService.verifyOrgMember(org_id, req.user.user_id) — return 403 if not member. Return 200 + org.

**PUT /api/v1/organizations/:org_id** — authenticate. UUID guard. IDOR: call verifyOrgAdmin(org_id, userId) — return 403 if not admin. Parse updateOrgSchema. Call organizationService.updateOrg. Return 200 + updated org.

**GET /api/v1/organizations/:org_id/credential-status** — authenticate. UUID guard. IDOR: verifyOrgMember → 403. Call organizationService.getCredentialStatus(org_id). Return 200 + CredentialStatus.

**GET /api/v1/organizations/:org_id/roles** — authenticate. UUID guard. IDOR: verifyOrgMember → 403. SELECT from org_roles WHERE org_id=$1 AND revoked_at IS NULL. Return 200 + array.

**POST /api/v1/organizations/:org_id/roles** — authenticate. UUID guard. Admin guard: verifyOrgAdmin → 403. Parse assignRoleSchema. INSERT into org_roles (org_id, user_id, roles, invited_by=req.user.user_id, invitation_sent_at=now()). On duplicate (uq_org_user_role), UPDATE roles WHERE revoked_at IS NULL. Return 201 + role row.

**PUT /api/v1/organizations/:org_id/roles/:role_id** — authenticate. UUID guard (both params). Admin guard. Parse updateRoleSchema. UPDATE org_roles SET roles=$1 WHERE role_id=$2 AND org_id=$3 AND revoked_at IS NULL. Return 200 + updated row.

**DELETE /api/v1/organizations/:org_id/roles/:role_id** — authenticate. UUID guard. Admin guard. Cannot revoke yourself (return 400 if role.user_id === req.user.user_id). UPDATE org_roles SET revoked_at=now() WHERE role_id=$1. Return 204.

**GET /api/v1/organizations/:org_id/documents** — authenticate. UUID guard. IDOR: verifyOrgMember → 403. Call organizationService.listDocuments(org_id). Return 200 + array.
